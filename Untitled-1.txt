import java.security.*;

class SecurePaymentSystem {
    private static final String SECRET_KEY = "1234567890123456"; // 16-char key
    private static final Map<String, Double> accountBalances = new ConcurrentHashMap<>();
    private static final List<String> transactionLogs = Collections.synchronizedList(new ArrayList<>());
    private static final Map<String, Long> transactionTimestamps = new ConcurrentHashMap<>();

    public static boolean isValidMastercard(String cardNumber) {
        if (!cardNumber.matches("5[1-5]\\d{14}")) return false;
        return passesLuhnAlgorithm(cardNumber);
    }

    private static boolean passesLuhnAlgorithm(String cardNumber) {
        int sum = 0;
        boolean alternate = false;

        for (int i = cardNumber.length() - 1; i >= 0; i--) {
            int digit = Character.getNumericValue(cardNumber.charAt(i));
            if (alternate) {
                digit *= 2;
                if (digit > 9) digit -= 9;
            }
            sum += digit;
            alternate = !alternate;
        }
        return (sum % 10 == 0);
    }

    public static String encryptCard(String cardNumber) throws Exception {
        Cipher cipher = Cipher.getInstance("AES");
        SecretKey key = new SecretKeySpec(SECRET_KEY.getBytes(), "AES");
        cipher.init(Cipher.ENCRYPT_MODE, key);
        byte[] encrypted = cipher.doFinal(cardNumber.getBytes());
        return Base64.getEncoder().encodeToString(encrypted);
    }

    public static String decryptCard(String encryptedCard) throws Exception {
        Cipher cipher = Cipher.getInstance("AES");
        SecretKey key = new SecretKeySpec(SECRET_KEY.getBytes(), "AES");
        cipher.init(Cipher.DECRYPT_MODE, key);
        byte[] decrypted = cipher.doFinal(Base64.getDecoder().decode(encryptedCard));
        return new String(decrypted);
    }

    public static boolean processTransaction(String cardNumber, double amount) throws Exception {
        String encryptedCard = encryptCard(cardNumber);

        if (!isValidMastercard(cardNumber)) {
            logTransaction(encryptedCard, "FAILED: Invalid Card");
            return false;
        }

        synchronized (accountBalances) {
            accountBalances.putIfAbsent(encryptedCard, 5000.0); // Default balance
            double balance = accountBalances.get(encryptedCard);
            if (amount > balance) {
                logTransaction(encryptedCard, "FAILED: Insufficient Funds");
                return false;
            }
            accountBalances.put(encryptedCard, balance - amount);
        }

        logTransaction(encryptedCard, "SUCCESS: Paid " + amount);
        return true;
    }

    public static void logTransaction(String card, String message) {
        String timestamp = new SimpleDateFormat("yyyy-MM-dd HH:mm:ss").format(new Date());
        String log = timestamp + " | Card: " + card + " | " + message;
        transactionLogs.add(log);
        System.out.println(log);
    }

    public static void detectFraud(String cardNumber) throws Exception {
        String encryptedCard = encryptCard(cardNumber);
        long now = System.currentTimeMillis();

        if (transactionTimestamps.containsKey(encryptedCard)) {
            long lastTransactionTime = transactionTimestamps.get(encryptedCard);
            if ((now - lastTransactionTime) < 5000) { // 5-second rule
                logTransaction(encryptedCard, "WARNING: Possible Fraud Detected!");
            }
        }

        transactionTimestamps.put(encryptedCard, now);
    }

    public static void printLogs() {
        System.out.println("\n=== Transaction Logs ===");
        for (String log : transactionLogs) {
            System.out.println(log);
        }
    }
